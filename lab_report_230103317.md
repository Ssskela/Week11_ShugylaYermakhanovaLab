### harness.py
```python
from dataclasses import dataclass
import multiprocessing as mp
import time
from typing import Callable


@dataclass
class BenchStats:
    min_ms: float
    median_ms: float
    iqr_ms: float


def fnv1a_hash(s: str) -> int:
    h = 14695981039346656037
    for b in s.encode("utf-8"):
        h = ((h ^ b) * 1099511628211) & 0xFFFFFFFFFFFFFFFF
    return h


def run_benchmark(fn: Callable[[], None], warmup: int = 2, runs: int = 7) -> BenchStats:
    for _ in range(warmup):
        fn()
    samples = []
    for _ in range(runs):
        t0 = time.perf_counter_ns()
        fn()
        samples.append((time.perf_counter_ns() - t0) / 1e6)
    samples.sort()
    med = samples[runs // 2]
    return BenchStats(samples[0], med, samples[(3 * runs) // 4] - samples[runs // 4])


def emit_verify_tag(phase: str, seed: int, m1: float, m2: float) -> str:
    hw = mp.cpu_count()
    sig = fnv1a_hash(f"{phase}|{seed}|{hw}|{m1:.4f}|{m2:.4f}")
    return f"VERIFY_HASH [{phase}]: {sig:016X} (P_log={hw})"
```

```python
            x = math.sin(x) * 1.0001 + math.cos(x)
        acc += x
    return acc


def run_pipeline(pool: mp.Pool, p: int, n_total: int) -> tuple[float, float]:
    t0 = time.perf_counter_ns()
    n_serial = int(0.05 * n_total)
    _ = stage_a_serial(n_serial)

    rem = n_total % p
    tasks = []
    offset = 0
    for idx in range(p):
        sz = base_chunk + (1 if idx < rem else 0)
        tasks.append((sz, offset))
        offset += sz

    pool.starmap(stage_b_worker, tasks)
    t_total = (time.perf_counter_ns() - t0) / 1e6
    return t_total, t_stage_a
```

```python
def flops_worker(iters: int) -> float:
    x0 = x1 = x2 = x3 = 1.0001
    x4 = x5 = x6 = x7 = 1.0002
    x8 = x9 = x10 = x11 = 1.0003
    x12 = x13 = x14 = x15 = 1.0004
    c1, c2 = 0.999999, 0.000001
    # 16 variables * 2 FLOPs (mul + add) = 32 FLOPs per loop iteration
    for _ in range(iters):
        x0 = x0 * c1 + c2; x1 = x1 * c1 + c2; x2 = x2 * c1 + c2; x3 = x3 * c1 + c2
        x4 = x4 * c1 + c2; x5 = x5 * c1 + c2; x6 = x6 * c1 + c2; x7 = x7 * c1 + c2
        x8 = x8 * c1 + c2; x9 = x9 * c1 + c2; x10 = x10 * c1 + c2; x11 = x11 * c1 + c2
        x12 = x12 * c1 + c2; x13 = x13 * c1 + c2; x14 = x14 * c1 + c2; x15 = x15 * c1 + c2
    return x0 + x15


def gemm_v1_worker(a_name: str, b_name: str, c_name: str, n: int, r0: int, r1: int) -> None:
    sa = shared_memory.SharedMemory(name=a_name)
    sb = shared_memory.SharedMemory(name=b_name)
    sc = shared_memory.SharedMemory(name=c_name)
    a = (ctypes.c_double * (n * n)).from_buffer(sa.buf)
    b = (ctypes.c_double * (n * n)).from_buffer(sb.buf)
    c = (ctypes.c_double * (n * n)).from_buffer(sc.buf)

    for i in range(r0, r1):
        for j in range(n):
            s = 0.0
            for k in range(n):
                s += a[i * n + k] * b[k * n + j]
            c[i * n + j] = s

    del a, b, c
    sa.close(); sb.close(); sc.close()

        for k in range(n):
            a_ik = a[i * n + k]
            row_b = k * n
            for j in range(n):
                c[row_c + j] += a_ik * b[row_b + j]

    del a, b, c
    sa.close(); sb.close(); sc.close()
```

```python
from harness import run_benchmark, emit_verify_tag

STUDENT_ID = 230103317
P_PHYS = 8  # Update to your laptop's NumberOfCores

def compute_item(i: int) -> float:
    k_iters = 1 + (i // 40) ** 2
    acc = 0.0
    x = (i + 1) * 0.001
    for _ in range(k_iters):
        x = math.sin(x) * math.cos(x) + math.sqrt(abs(x) + 1.0)
        acc += x

def static_block_worker(tid: int, p: int, n: int, t_arr) -> None:
    t0 = time.perf_counter_ns()
    chunk = n // p
    start = tid * chunk
    end = n if tid == p - 1 else (tid + 1) * chunk
    for i in range(start, end):
        compute_item(i)
    t_arr[tid] = (time.perf_counter_ns() - t0) / 1e6


def static_cyclic_worker(tid: int, p: int, n: int, t_arr) -> None:
    t0 = time.perf_counter_ns()
    for i in range(tid, n, p):
        compute_item(i)
    t_arr[tid] = (time.perf_counter_ns() - t0) / 1e6


def dynamic_queue_worker(tid: int, n: int, chunk_size: int, counter, t_arr) -> None:
    t0 = time.perf_counter_ns()
    while True:
        with counter.get_lock():
            start = counter.value
            if start >= n:
                break
            counter.value = start + chunk_size
        end = min(start + chunk_size, n)
        for i in range(start, end):
            compute_item(i)
    t_arr[tid] = (time.perf_counter_ns() - t0) / 1e6


def eval_strategy(mode: str, chunk_size: int = 1) -> tuple[float, float]:
    t_arr = mp.Array("d", P_PHYS)
    counter = mp.Value("i", 0)

    def run_once():
        counter.value = 0
        procs = []
        for tid in range(P_PHYS):
            if mode == "block":
                p = mp.Process(target=static_block_worker, args=(tid, P_PHYS, N_ITEMS, t_arr))
            elif mode == "cyclic":
                p = mp.Process(target=static_cyclic_worker, args=(tid, P_PHYS, N_ITEMS, t_arr))
            else:
                p = mp.Process(target=dynamic_queue_worker, args=(tid, N_ITEMS, chunk_size, counter, t_arr))
            procs.append(p)
        for p in procs:
            p.start()
        for p in procs:
            p.join()

    stats = run_benchmark(run_once, warmup=1, runs=5)
    run_once()
    times = [t_arr[i] for i in range(P_PHYS)]
    imbalance = max(times) / (sum(times) / P_PHYS)
    return stats.median_ms, imbalance


def main():
    print("=== Phase 4: Skewed Workloads & Dynamic Chunk Queue ===")
    t_block, lam_block = eval_strategy("block")
```
